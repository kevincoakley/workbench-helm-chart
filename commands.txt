2.0 - Production
================
# Make sure all of the helm dependencies are up to date
helm dep up

# Create a namespace for the dev environment
kubectl create namespace workbench

# Add add the SSL cert to the dev environment
kubectl -n workbench create secret tls workbench-tls \
  --cert=kc-workbench.cer \
  --key=kc-workbench.key

# Create the workbench pv and pvc
kubectl apply -f kc-workbench-mongodb.yaml
kubectl apply -f kc-workbench-nfs.yaml

# Install the workbench chart
helm upgrade --install workbench . \
  --namespace workbench --create-namespace \
  -f kc-gke-prod-values.yaml

# Delete the workbench chart
helm -n workbench uninstall workbench

# Delete the workbench pv and pvc
kubectl delete -f kc-workbench-mongodb.yaml
kubectl delete -f kc-workbench-nfs.yaml

# Delete the namespace for the environment
kubectl delete namespace workbench
